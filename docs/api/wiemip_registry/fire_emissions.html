wiemip_registry.fire_emissions

Fire CH4, N2O and CO2 emissions (kg gas m-2 s-1) for any overshoot model.

import wiemip_registry as wr
ch4 = wr.fire_emission('CLM', 'hist', 'CH4', period=('1994', '2023'))
n2o = wr.fire_emission('JSBACH', 'hl', 'N2O', forcing='ipsl')
ch4.attrs['source']   # 'model', 'NEIVA pft_flux', 'NEIVA cveg_pft' or 'NEIVA latitude'

Returns the model's own output if it exists, otherwise fFire converted with NEIVA emission ratios. The result is an xr.DataArray on the model's grid and time axis; aggregation (monthly/annual, regional, global) is up to the user.

With method=None the first possible step is used (a model with neiva_method in MODEL_SETTINGS uses that NEIVA method for every run instead of steps 2-4):

  1. model : fCH4Fire / fN2OFire, units converted from the units attribute
  2. pft_flux : fFire * sum_pft fFirepft/sum(fFirepft) * f(pft) (fFirepft + PFT table)
  3. cveg_pft : fFire * sum_pft cVegpft/cVeg_tot * f(pft) (cVegpft + PFT table)
  4. latitude : fFire * f(fire type), forest where the hist mean cVeg over ref_period > 5 kg C m-2
Fire type Criterion
tropical forest forest, abs(lat) < 30
temperate forest forest, 30 <= abs(lat) < 60
boreal forest abs(lat) >= 60
savanna non-forest, abs(lat) < 60

Emission = fFire x NEIVA ratio of the fire type x molar mass ratio (CH4 16.04/12.011, N2O 44.013/28.014, CO2 44.009/12.011).

What fFire contains is set per model in MODEL_SETTINGS (ffire_basis):

ffire_basis fFire is ratios used CO2
'C' (default) all emitted carbon f_CH4-C/C, f_N2O-N/C fFire x f_CO2-C/C x 44/12
'CO2' carbon emitted as CO2 f_CH4-C/C-CO2, f_N2O-N/C-CO2 fFire x 44/12

neiva_pft_map.csv (next to this module, model;pft;neiva;comment) maps each model PFT to a NEIVA fire type, forest / nonforest (type from latitude) or none. pft is the PFT label as read() returns it, or the position 1..n for numbered PFTs. Every PFT of the model needs a row. Per-PFT fields in the adapter's PER_PFT_AREA are weighted by the run's landCoverFrac.

Missing NEIVA N2O values are taken from a similar fire type (tropical -> temperate forest; peat -> boreal forest; crop, pasture -> savanna). With fixed ratios per fire type, relative changes of CH4, N2O and CO2 follow fFire; the gases differ only where fire shifts between types.

 57# ---------------------------------------------------------------- NEIVA table
 58#   f_CH4-C/C, f_N2O-N/C         : per kg C emitted in total
 59#   f_CH4-C/C-CO2, f_N2O-N/C-CO2 : per kg C emitted as CO2
 60# N2O is missing in NEIVA for tropical_forest, crop_residue, peat, pasture_maintenance
 61# -> taken from temperate_forest, savanna, boreal_forest, savanna.
 62#                        f_CH4-C/C  f_N2O-N/C  f_CH4-C/C-CO2  f_N2O-N/C-CO2
 63TABLE = {
 64    "savanna": (0.004105, 0.000173, 0.004730, 0.000199),
 65    "boreal_forest": (0.006975, 0.000254, 0.008153, 0.000297),
 66    "tropical_forest": (0.006826, 0.000191, 0.007905, 0.000229),
 67    "temperate_forest": (0.006874, 0.000191, 0.008233, 0.000229),
 68    "peat": (0.013890, 0.000254, 0.019372, 0.000297),
 69    "chaparral": (0.003805, 0.000315, 0.004279, 0.000354),
 70    "crop_residue": (0.003537, 0.000173, 0.004078, 0.000199),
 71    "pasture_maintenance": (0.012847, 0.000173, 0.015688, 0.000199),
 72}
 73# Factors applied to fFire, depending on what fFire contains:
 74#   'C'   : all emitted carbon    -> /C columns,     CO2-C/C = f_CH4-C/C / f_CH4-C/C-CO2
 75#   'CO2' : carbon emitted as CO2 -> /C-CO2 columns, CO2-C/fFire = 1
 76F = {
 77    "C": {t: {"CH4": a, "N2O": b, "CO2": a / c} for t, (a, b, c, d) in TABLE.items()},
 78    "CO2": {t: {"CH4": c, "N2O": d, "CO2": 1.0} for t, (a, b, c, d) in TABLE.items()},
 79}
 80M = {
 81    "CH4": 16.04 / 12.011,  # kg C -> kg CH4
 82    "N2O": 44.013 / 28.014,  # kg N -> kg N2O
 83    "CO2": 44.009 / 12.011,
 84}  # kg C -> kg CO2
 85UNITS = {g: f"kg {g} m-2 s-1" for g in M}
 86FOREST_THRESHOLD = 5.0  # kg C m-2

 90#   ffire_basis  : 'C' fFire = all emitted carbon | 'CO2' fFire = CO2-C only
 91#   model_factor : extra factor for the model's own gas output, per gas - only needed if

 99    ffire_basis="C", model_factor={}, use_pft_flux=True, retrieve={}, neiva_method=None

108    # (fire_n2o_derivation_README.md: 0.5 kg C per kg DM, CO 100 g and CH4 6.8 g per kg DM)

116def _fire_in_kgC(da):
117    """fFire / fFirePFT in kg C m-2 s-1; converts if the units say kg CO2 (mass of CO2)."""
118    if da is None:
119        return None
120    units = da.attrs.get("units") or ""
121    u = units.replace("kg", " kg ").split()
122    elem = u[u.index("kg") + 1] if "kg" in u and u.index("kg") + 1 < len(u) else None
123    if elem == "CO2":
124        return (da * (1 / M["CO2"])).assign_attrs(units="kg C m-2 s-1")
125    if elem != "C":
126        warnings.warn(
127            f"fire flux units {units!r} not recognised - used as kg C m-2 s-1"
128        )
129    return da

132def _unit_factor(units, gas):
133    """Factor to kg gas from the units attribute: 'kg N ...' -> x M['N2O'],
134    'kg C ...' -> x M['CH4'] / M['CO2']; 'kg <gas> ...' -> 1."""
135    u = (units or "").replace("kg", " kg ").split()
136    if "kg" not in u or u.index("kg") + 1 >= len(u):
137        warnings.warn(
138            f"f{gas}Fire: units {units!r} not recognised - used as kg {gas} m-2 s-1"
139        )
140        return 1
141    elem = u[u.index("kg") + 1]
142    if elem == gas:
143        return 1
144    if (gas, elem) in {("N2O", "N"), ("CH4", "C"), ("CO2", "C")}:
145        return M[gas]
146    warnings.warn(
147        f"f{gas}Fire: units {units!r} not recognised - used as kg {gas} m-2 s-1"
148    )
149    return 1

152# PFT -> NEIVA table: columns model, pft, neiva (optional: comment).
153# neiva = a NEIVA fire type from TABLE, 'forest' / 'nonforest' (type from latitude)
154# or 'none' (no fire). pft = PFT name as in the model files, or for numbered PFTs the
155# position 1..n in file order (whether the file counts from 0 or from 1).
156PFT_TABLE = Path(__file__).with_name("neiva_pft_map.csv")
157TOKENS = {"forest", "nonforest"}

160def pft_map_for(model, table=PFT_TABLE):
161    """PFT label -> NEIVA type for one model from the central table ({} if none)."""
162    if not Path(table).exists():
163        return {}
164    # separator from the header line (',' or ';' - e.g. Excel with German settings)
165    with open(table, encoding="utf-8-sig") as fh:
166        sep = ";" if ";" in fh.readline() else ","
167    df = pd.read_csv(table, sep=sep, dtype=str, encoding="utf-8-sig").fillna("")
168    df.columns = df.columns.str.strip().str.lower()
169    out = {}
170    for r in df[df["model"].str.strip() == model].itertuples():
171        v = r.neiva.strip()
172        if v.lower() in ("", "none"):
173            v = None
174        elif v not in TABLE and v not in TOKENS:
175            raise ValueError(
176                f"{table.name}: unknown NEIVA type {v!r} for {model} / {r.pft}"
177            )
178        out[r.pft.strip()] = v
179    return out

182# ---------------------------------------------------------------- conversion
183def neiva(
184    gas,
185    fFire=None,
186    fFirePFT=None,
187    cVegPFT=None,
188    cveg_ref=None,
189    pft_map=None,
190    landCoverFrac=None,
191    ffire_basis="C",
192):
193    """
194    NEIVA conversion -> kg gas m-2 s-1.
195    Give fFirePFT (+ pft_map), or fFire with cVegPFT (+ pft_map) or with cveg_ref
196    (time-mean cVeg, kg C m-2, same grid). Fluxes in kg C m-2 s-1.
197    landCoverFrac (fraction) is applied to per-PFT fields if given.
198    ffire_basis: 'C' if fFire is all emitted carbon, 'CO2' if it is CO2-carbon only.
199    """
200    Fb = F[ffire_basis]
201    if fFirePFT is not None:
202        E, d = fFirePFT, _pft_dim(fFirePFT)
203        f = _pft_factor(E[d], E["lat"], gas, pft_map, Fb)
204        lcf = (
205            _grid(_match_pft(landCoverFrac, E), E)
206            if landCoverFrac is not None
207            else None
208        )
209        num = den = valid = None
210        for k in range(E.sizes[d]):
211            e = E.isel({d: k}, drop=True)
212            if lcf is not None:
213                e = e * _to_time(lcf.isel({d: k}, drop=True), e)
214            ok = e.notnull()
215            e = e.fillna(0)
216            t = e * f.isel({d: k}, drop=True)
217            num = t if num is None else num + t
218            den = e if den is None else den + e
219            valid = ok if valid is None else (valid | ok)
220            del e, t, ok
221        if fFire is not None:
222            eff = (num / den.where(den > 0)).fillna(
223                _lat_factor(E["lat"], gas, False, Fb)
224            )
225            out = fFire * _to_time(_grid(eff.where(valid), fFire), fFire)
226        else:
227            out = num.where(valid)
228        method = "pft_flux"
229    elif cVegPFT is not None:
230        C, d = cVegPFT, _pft_dim(cVegPFT)
231        if landCoverFrac is not None:
232            C = C * _to_time(_grid(_match_pft(landCoverFrac, C), C), C)
233        tot = C.sum(d)
234        f = (
235            ((C / tot.where(tot > 0)) * _pft_factor(C[d], C["lat"], gas, pft_map, Fb))
236            .sum(d, min_count=1)
237            .fillna(_lat_factor(C["lat"], gas, False, Fb))
238        )
239        out = fFire * _to_time(_grid(f, fFire), fFire)
240        method = "cveg_pft"
241    else:
242        cveg_ref = _grid(cveg_ref, fFire)
243        f = xr.where(
244            cveg_ref > FOREST_THRESHOLD,
245            _lat_factor(fFire["lat"], gas, True, Fb),
246            _lat_factor(fFire["lat"], gas, False, Fb),
247        )
248        out = fFire * f.where(cveg_ref.notnull())
249        method = "latitude"
250    emission = (out * M[gas]).rename(f"f{gas}Fire")
251    emission.attrs = dict(
252        long_name=f"{gas} emission from fire (fFire x NEIVA emission ratio)",
253        units=UNITS[gas],
254        source=f"NEIVA {method}",
255        ffire_basis=ffire_basis,
256    )
257    return emission

260def _lat_factor(lat, gas, forest, Fb):
261    """
262    NEIVA factor by latitude band:
263      forest     : |lat| < 30 tropical_forest, 30-60 temperate_forest, >= 60 boreal_forest
264      non-forest : |lat| < 60 savanna,                                  >= 60 boreal_forest
265    """
266    a = abs(lat)
267    if forest:
268        return xr.where(
269            a < 30,
270            Fb["tropical_forest"][gas],
271            xr.where(a < 60, Fb["temperate_forest"][gas], Fb["boreal_forest"][gas]),
272        )
273    return xr.where(a < 60, Fb["savanna"][gas], Fb["boreal_forest"][gas])

276def _key(p):
277    p = p.decode() if isinstance(p, bytes) else (p.item() if hasattr(p, "item") else p)
278    if isinstance(p, float) and p.is_integer():
279        p = int(p)  # 1.0 -> 1 (PFT labels stored as float)
280    return str(p).strip()

283def _pft_keys(labels):
284    """PFT keys for the table: names as they are; numeric labels -> position 1..n
285    in file order (files of the same model may count from 0 or from 1)."""
286    keys = [_key(p) for p in labels.values]
287    if all(k.lstrip("-").isdigit() for k in keys):
288        return [str(i + 1) for i in range(len(keys))]
289    return keys
290
291
292def _pft_factor(labels, lat, gas, pft_map, Fb):
293    pft_map = {str(k).strip(): v for k, v in pft_map.items()}
294    keys = _pft_keys(labels)
295    missing = [k for k in keys if k not in pft_map]
296    if missing:
297        raise KeyError(
298            f"PFTs not in the PFT table (numeric PFTs = position 1..n): {missing}"
299        )
300    fs = []
301    for k in keys:
302        t = pft_map[k]
303        if t is None:
304            fs.append(xr.zeros_like(lat, dtype=float))
305        elif t in TOKENS:
306            fs.append(_lat_factor(lat, gas, t == "forest", Fb))
307        else:
308            fs.append(xr.full_like(lat, Fb[t][gas], dtype=float))
309    return xr.concat(fs, labels)

312def _pft_dim(da):
313    extra = [d for d in da.dims if d not in ("time", "lat", "lon")]
314    if len(extra) != 1:
315        raise ValueError(f"cannot identify the PFT dimension in {da.dims}")
316    return extra[0]

319def _match_pft(lcf, ref):
320    """Give landCoverFrac the same PFT dimension name and labels as ref."""
321    d, dr = _pft_dim(lcf), _pft_dim(ref)
322    if lcf.sizes[d] != ref.sizes[dr]:
323        raise ValueError(
324            f"landCoverFrac has {lcf.sizes[d]} PFTs, field has {ref.sizes[dr]}"
325        )
326    return (
327        lcf.rename({d: dr}).assign_coords({dr: ref[dr]})
328        if d != dr
329        else lcf.assign_coords({dr: ref[dr]})
330    )

333def _grid(da, ref):
334    return da.assign_coords(lat=ref["lat"], lon=ref["lon"])

337def _to_time(da, ref):
338    """Put da on ref's time axis (annual -> monthly by year)."""
339    if "time" not in da.dims:
340        return da
341    if da.sizes["time"] != ref.sizes["time"]:
342        da = (
343            da.groupby("time.year")
344            .mean()
345            .sel(year=ref["time"].dt.year, method="nearest")
346            .drop_vars("year")
347        )
348    return da.assign_coords(time=ref["time"])

367def _load(experiment, model, forcing, sim, factorial, name, period=None, retrieve=None):
368    """(DataArray, WIEFile) for one variable; (None, None) if it cannot be read."""
369    try:
370        if experiment == ONE_PCT:
371            wiefile = wiemip_registry.retrieve_one_pct_variable(
372                model=model,
373                forcing=forcing,
374                simulation=sim,
375                factorial=factorial or "baseline",
376                variable=name,
377            )
378        else:
379            kwargs = dict(retrieve or {})
380            if factorial is not None:
381                kwargs["factorial"] = factorial
382            wiefile = wiemip_registry.retrieve_overshoot_variable(
383                model=model, forcing=forcing, simulation=sim, variable=name, **kwargs
384            )
385        da = wiefile.read()
386    except NOT_AVAILABLE as e:
387        if VERBOSE:
388            print(
389                f"[fire_emission] {experiment} {model} {forcing} {sim} {factorial} "
390                f"{name}: not available ({type(e).__name__}: {e})"
391            )
392        return None, None
393    except Exception as e:
394        warnings.warn(
395            f"{model} {sim} {name}: read failed "
396            f"({type(e).__name__}: {e}) - treated as not available"
397        )
398        return None, None
399    if period:  # by year: works beyond 2262 too
400        y = da["time.year"]
401        da = da.sel(time=(y >= int(period[0])) & (y <= int(period[1])))
402    return da, wiefile

405def _pft_area_fraction(da, wiefile):
406    """Share of the registry's cell weight each PFT covers, for a PER_PFT_AREA variable."""
407    if wiefile.variable not in wiefile._adapter.PER_PFT_AREA:
408        return None
409    cell_weights = wiefile._adapter.weights()
410    pft_area = wiefile._pft_area_weights(da)
411    return pft_area / cell_weights.where(cell_weights > 0)

194    """
195    NEIVA conversion -> kg gas m-2 s-1.
196    Give fFirePFT (+ pft_map), or fFire with cVegPFT (+ pft_map) or with cveg_ref
197    (time-mean cVeg, kg C m-2, same grid). Fluxes in kg C m-2 s-1.
198    landCoverFrac (fraction) is applied to per-PFT fields if given.
199    ffire_basis: 'C' if fFire is all emitted carbon, 'CO2' if it is CO2-carbon only.
200    """
201    Fb = F[ffire_basis]
202    if fFirePFT is not None:
203        E, d = fFirePFT, _pft_dim(fFirePFT)
204        f = _pft_factor(E[d], E["lat"], gas, pft_map, Fb)
205        lcf = (
206            _grid(_match_pft(landCoverFrac, E), E)
207            if landCoverFrac is not None
208            else None
209        )
210        num = den = valid = None
211        for k in range(E.sizes[d]):
212            e = E.isel({d: k}, drop=True)
213            if lcf is not None:
214                e = e * _to_time(lcf.isel({d: k}, drop=True), e)
215            ok = e.notnull()
216            e = e.fillna(0)
217            t = e * f.isel({d: k}, drop=True)
218            num = t if num is None else num + t
219            den = e if den is None else den + e
220            valid = ok if valid is None else (valid | ok)
221            del e, t, ok
222        if fFire is not None:
223            eff = (num / den.where(den > 0)).fillna(
224                _lat_factor(E["lat"], gas, False, Fb)
225            )
226            out = fFire * _to_time(_grid(eff.where(valid), fFire), fFire)
227        else:
228            out = num.where(valid)
229        method = "pft_flux"
230    elif cVegPFT is not None:
231        C, d = cVegPFT, _pft_dim(cVegPFT)
232        if landCoverFrac is not None:
233            C = C * _to_time(_grid(_match_pft(landCoverFrac, C), C), C)
234        tot = C.sum(d)
235        f = (
236            ((C / tot.where(tot > 0)) * _pft_factor(C[d], C["lat"], gas, pft_map, Fb))
237            .sum(d, min_count=1)
238            .fillna(_lat_factor(C["lat"], gas, False, Fb))
239        )
240        out = fFire * _to_time(_grid(f, fFire), fFire)
241        method = "cveg_pft"
242    else:
243        cveg_ref = _grid(cveg_ref, fFire)
244        f = xr.where(
245            cveg_ref > FOREST_THRESHOLD,
246            _lat_factor(fFire["lat"], gas, True, Fb),
247            _lat_factor(fFire["lat"], gas, False, Fb),
248        )
249        out = fFire * f.where(cveg_ref.notnull())
250        method = "latitude"
251    emission = (out * M[gas]).rename(f"f{gas}Fire")
252    emission.attrs = dict(
253        long_name=f"{gas} emission from fire (fFire x NEIVA emission ratio)",
254        units=UNITS[gas],
255        source=f"NEIVA {method}",
256        ffire_basis=ffire_basis,
257    )
258    return emission
