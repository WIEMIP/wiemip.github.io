wiemip_registry.core

Core of wiemip_registry: the WIEAdapter contract every per-model adapter fills out, the WIEFile wrapper the dotted namespace resolves to, and the small set of generic helpers the adapters share.

Per-model knowledge (paths, dims, time encoding, area recipe, fills) lives in each <MODEL>/convert.py as a WIEAdapter subclass. This module only holds the generic mechanics (spherical area, standardize, fill masking, weighted aggregation), all seeded from the proven extract.py.

 70def ensure_valid(method):
 71    """
 72    Hand-deny any strange combination of arguments.
 73    Each adapter can have pathological combinations and decide whether to
 74    accept unconventional arguments (e.g., UKESM forcing a ctrl simulation),
 75    so we short-circuit that here.
 76    So:
 77     bgc/ctrl should only return True for stable climate
 78     cou should only return True for ukesm/gfdl/ipsl
 79     otherwise we should return false
 80    """
 81
 82    @functools.wraps(method)
 83    def wrapper(self):
 84
 85        if self.experiment == const.ONE_PERCENT_CO2_KEY:
 86            # cou and rad both include the radiative effect of CO2, so they are
 87            # driven with a transient GCM-pattern climate. bgc and ctrl hold the
 88            # climate constant, so they take the stable pattern.
 89            if self.simulation in (
 90                const.OnePctSimulation.cou.name,
 91                const.OnePctSimulation.cou_ndep.name,
 92                const.OnePctSimulation.rad.name,
 93                const.OnePctSimulation.rad_ndep.name,
 94            ):
 95                if self.forcing not in (
 96                    const.GCMPattern.ukesm.name,
 97                    const.GCMPattern.ipsl.name,
 98                    const.GCMPattern.gfdl.name,
 99                ):
100                    self.invalid_combination = True
101                    return False
102            elif self.simulation in (
103                const.OnePctSimulation.bgc.name,
104                const.OnePctSimulation.bgc_ndep.name,
105            ):
106                if self.forcing != const.GCMPattern.stable.name:
107                    self.invalid_combination = True
108                    return False
109            elif self.simulation == const.OnePctSimulation.ctrl.name:
110                if self.forcing != const.GCMPattern.stable.name:
111                    self.invalid_combination = True
112                    return False
113
114        return method(self)
115
116    return wrapper

141class WIEAdapter(ABC):
142    """
143    Contract that each model must fill out. This converts whatever naming
144    convention was uploaded to the WIE-MIP S3 bucket into something callable in
145    Python.
146
147    Subclasses implement the three abstract hooks (`path`, `read`,
148    `_compute_weights`). There is no per-model "what exists" schema: the namespace
149    lets a user select any (experiment, model, forcing, simulation, factorial,
150    variable) by name, and a combination that wasn't uploaded simply fails when
151    `read()` tries to open the file.
152    """
153
154    model: str
155
156    _weights_cache: xr.DataArray | None = None
157
158    # Per-model factorial vocabulary: canonical bucket -> however THIS model spells
159    # the factorial
160    # either overriden or set in the adapter subclass
161    FACTORIALS: dict[str, str] = {"baseline": ""}
162
163    # Same idea for the overshoot arm, empty unless the group ran more than one
164    # overshoot configuration (only JULES did).
165    OVERSHOOT_FACTORIALS: dict[str, str] = {}
166
167    def land_carbon_variables(self) -> list[str]:
168        """The variables that make up each model's land carbon stock. Usually some combination
169        of cVeg, cSoil, cLitter, and cOther.
170        """
171        raise NotImplementedError()
172
173    @property
174    def factorials(self) -> tuple[str, ...]:
175        """The factorial names this model accepts (drives namespace validation
176        and tab-completion of the factorial axis)."""
177        return tuple(self.FACTORIALS)
178
179    @abc.abstractmethod
180    def one_pct_path(
181        self,
182        simulation: str,
183        forcing: str,
184        factorial: str,
185        variable: str,
186    ) -> str:
187        """Build the .nc path on the mounted S3 bucket by transforming the axis
188        tokens into THIS model's upload naming convention."""
189
190        raise NotImplementedError()
191
192    def overshoot_path(
193        self,
194        simulation: str,
195        forcing: str,
196        variable: str,
197        factorial: str | None = None,
198    ) -> str:
199        """Overshoot-experiment path. Overridden per model once that model's overshoot
200        upload naming is known; until then asking for an overshoot path raises here
201        rather than guessing a layout. Like `one_pct_path` it's a pure string transform
202        — what exists is decided by `read()`.
203
204        Most groups ran a single overshoot configuration, so `factorial` is None by
205        default and those adapters ignore it. JULES is the exception: it repeated the
206        whole scenario set under five fire configs, so the factorial picks the run."""
207        raise NotImplementedError(f"overshoot paths not yet mapped for {self.model}")
208
209    def paths(self, experiment, simulation, forcing, factorial, variable) -> list[str]:
210        """Every file that makes up one variable's series, in time order. One file per
211        variable for everyone except CLM's overshoot upload, which splits each run into
212        time chunks, so the default just wraps `path`."""
213        return [self.path(experiment, simulation, forcing, factorial, variable)]
214
215    def path(self, experiment, simulation, forcing, factorial, variable):
216        """Return the expected path of the file defined by the experiment, simulation, forcing, factorial, and variable.
217        This is not guaranteed to exist - it just constructs the path.
218        """
219
220        if experiment == const.ONE_PERCENT_CO2_KEY:
221            if factorial not in self.FACTORIALS:
222                raise MissingFactorialError(
223                    f"{self.model} has no '{factorial}' factorial (has: {sorted(self.FACTORIALS)})"
224                )
225            pth = self.one_pct_path(simulation, forcing, factorial, variable)
226        elif experiment == "overshoot":
227            pth = self.overshoot_path(simulation, forcing, variable, factorial)
228        else:
229            raise ValueError("Must specify either overshoot or one_percent_co2!")
230        return pth
231
232    @abc.abstractmethod
233    def read(
234        self,
235        experiment: str,
236        simulation: str,
237        forcing: str,
238        factorial: str,
239        variable: str,
240    ) -> xr.DataArray:
241        """
242        Open one variable and STANDARDIZE its layout: canonical dims
243        ('time', 'lat', 'lon'[, level]), pd.DateTime `time` coord, sentinel fills
244        masked to NaN. Units stay NATIVE — unit conversion happens in `WIEFile`,
245        not here. Returns an unweighted DataArray.
246        """
247        raise NotImplementedError()
248
249    @abc.abstractmethod
250    def _compute_weights(self) -> xr.DataArray:
251        """Grid-cell weight [m2] for this model, per its README (provided raster
252        OR computed spherical area), standardized to ('lat', 'lon')."""
253        raise NotImplementedError()
254
255    def weights(self) -> xr.DataArray:
256        """Model weights [m2], materialized once and cached on the instance.
257        Fills are zeroed: a land-only / ocean-masked area raster carries NaN over
258        the cells it excludes, and `xarray.weighted()` rejects NaN weights — a
259        zero weight drops the cell from the integral, which is what we want."""
260        if self._weights_cache is None:
261            self._weights_cache = self._compute_weights().fillna(0.0)
262        return self._weights_cache
263
264    def weight_dataarray(self, da: xr.DataArray) -> xr.core.weighted.DataArrayWeighted:
265        """Wrap `da` in this model's documented weights via xarray `.weighted()`,
266        so `.sum()`/`.mean()` over (lat, lon) are one call."""
267        return da.weighted(self.weights())
268
269    def to_pgc(self, total: xr.DataArray, variable: str) -> pd.Series:
270        """Convert a per-timestep weighted (lat, lon) sum to a Pg C series,
271        PRESERVING the file's native cadence (monthly stays monthly — no annual
272        collapse). MODEL-SPECIFIC: override when a model's upload units make the
273        default wrong (e.g. its flux is not a per-second rate that SPY applies to).
274        Default: stock `/PG`; flux rate `×SPY/PG` (a per-timestep annualized rate)."""
275        s = total.to_series()
276        s = s / const.PG if kind_of(variable) == "stock" else s * const.SPY / const.PG
277        s.name = variable
278        return s
279

281def kind_of(variable: str) -> str:
282    """'stock' or 'flux' — selects the global-integral unit conversion."""
283    return "stock" if variable in const.STOCKS else "flux"

293def spherical_area(
294    obj: xr.Dataset | xr.DataArray, latn: str, lonn: str
295) -> xr.DataArray:
296    """
297    Grid-cell area [m2] from lat/lon centres, assuming a regular spherical grid.
298
299    Used by the models that do *not* ship an area raster (CLASSIC, DLEM, JSBACH,
300    JULES cell, VISIT-UT). Models with a provided raster (BiomeE `veg_area.nc`,
301    LPX-Bern `gridcell_area.nc`) ignore this. Verbatim from extract.py.
302    """
303    lat, lon = obj[latn].values, obj[lonn].values
304    R = 6.371e6
305    dlat, dlon = np.abs(np.gradient(lat)), np.abs(np.gradient(lon))
306    band = R**2 * (
307        np.sin(np.deg2rad(lat + dlat / 2)) - np.sin(np.deg2rad(lat - dlat / 2))
308    )
309    return xr.DataArray(
310        band[:, None] * np.deg2rad(dlon)[None, :],
311        dims=(latn, lonn),
312        coords={latn: obj[latn], lonn: obj[lonn]},
313    ).astype("float32")

321def rename_latlon(da: xr.DataArray, latn: str, lonn: str) -> xr.DataArray:
322    """Rename a model's native lat/lon dims to the canonical ('lat', 'lon')."""
323    ren = {}
324    if latn != "lat":
325        ren[latn] = "lat"
326    if lonn != "lon":
327        ren[lonn] = "lon"
328    return da.rename(ren) if ren else da

331def standardize(
332    da: xr.DataArray, latn: str, lonn: str, time: np.ndarray
333) -> xr.DataArray:
334    """
335    Map a model's raw DataArray onto the canonical standardized form: lat/lon
336    renamed to ('lat', 'lon'), a pandas-`datetime64` `time` coord attached
337    (PRESERVING the file's native cadence — monthly stays monthly), and
338    ('time', 'lat', 'lon') moved to the front while keeping any extra dims
339    (e.g. PFT / soil levels). `da` is fill-masked already; `time` is the decoded
340    datetime axis from the adapter's `_time(ds)` hook.
341    """
342    da = rename_latlon(da, latn, lonn)
343    da = da.assign_coords(time=("time", to_datetime64(time)))
344    front = [d for d in ("time", "lat", "lon") if d in da.dims]
345    rest = [d for d in da.dims if d not in front]
346    return da.transpose(*front, *rest)

349def to_datetime64(values) -> np.ndarray:
350    """A decoded time axis -> `datetime64[us]`.
351
352    Microseconds, not nanoseconds: the overshoot runs end in 2300 and datetime64[ns]
353    stops at 2262, so an ns cast silently wrapped every overshoot series past that year
354    (2299 came back as 1714). Files that far out decode to cftime objects rather than
355    datetime64, so pull the fields off them directly.
356    """
357    arr = np.asarray(values)
358    if arr.dtype == object:  # cftime.datetime, one per timestep
359        arr = np.array(
360            [
361                f"{t.year:04d}-{t.month:02d}-{t.day:02d}T{t.hour:02d}:{t.minute:02d}"
362                for t in arr
363            ],
364            dtype="datetime64[us]",
365        )
366    return arr.astype("datetime64[us]")

369def years_to_datetime(values) -> np.ndarray:
370    """Numeric (possibly fractional) *calendar* years -> `datetime64[M]`, keeping
371    sub-annual resolution: year = floor(v), month = round(frac * 12) clamped 0..11.
372    Annual data (frac == 0) maps to January of each year. Used by the models whose
373    time axis is a bare numeric year (LPX-Bern, VISIT-UT)."""
374    v = np.asarray(values, dtype="float64")
375    years = np.floor(v).astype("int64")
376    months = np.clip(np.rint((v - years) * 12).astype("int64"), 0, 11)
377    total_months = (years - 1970) * 12 + months  # months since the 1970 epoch
378    return np.datetime64("1970-01", "M") + total_months.astype("timedelta64[M]")

381def _csv_path(src: Path, start: float | None, end: float | None) -> Path:
382    """Cache path for a source .nc + latitude band: the source path mirrored under
383    `const.CSV_ROOT` (the `csv/` prefix), '.nc' -> '.csv', suffixed with the band
384    ('global' for whole-globe, else '<start>_<end>')."""
385    band = "global" if start is None or end is None else f"{start}_{end}"
386    rel = src.relative_to(const.DATA_ROOT)
387    return const.CSV_ROOT / rel.parent / f"{rel.stem}_{band}.csv"

390def cache_csv(method):
391    """Lazy CSV cache for a `WIEFile` (lat, lon)->time aggregation returning a
392    `pd.Series`. Mirrors the result to a CSV under `const.CSV_ROOT` and recomputes
393    only when that CSV is missing or older than the source variable file; on a hit
394    it reads the CSV straight back (two stats, zero netCDF reads).
395
396    Freshness keys on the *variable* file (`self.path`) only.
397    Use overwrite to recompute the sum. Useful when cache is invalid or methods change.
398    """
399
400    @functools.wraps(method)
401    def wrapper(self, start=None, end=None, overwrite=False):
402        srcs = [Path(p) for p in self.paths]  # pure transform == what read() opens
403        out = _csv_path(srcs[0], start, end)
404        if (
405            not overwrite
406            and out.exists()
407            and all(src.exists() for src in srcs)
408            and out.stat().st_mtime >= max(src.stat().st_mtime for src in srcs)
409        ):
410            series = pd.read_csv(out, index_col=0, parse_dates=True).iloc[:, 0]
411            # read_csv picks a resolution per file, so a series that stops before 2262
412            # comes back in ns and one running to 2300 in us; adding the two then
413            # overflows. Pin both to what standardize() hands out.
414            series.index = series.index.astype("datetime64[us]")
415            return series
416        series = method(self, start, end)
417        out.parent.mkdir(parents=True, exist_ok=True)
418        series.to_csv(out)
419        return series
420
421    return wrapper

424@dataclass
425class WIEFile:
426    """
427    Thin, lazy wrapper over one variable's file(s) for one run. Holds identity +
428    the model's adapter instance. No s3 access until a data method is called.
429    """
430
431    model: str  # canonical model name, e.g. "LPX-Bern"
432    experiment: str  # on-disk experiment dir, "1pctCO2" | "overshoot"
433    simulation: str  # per-experiment run name, e.g. "bgc", "hl"
434    forcing: str  # GCM pattern name, e.g. "ukesm"
435    variable: str  # CMIP name, e.g. "cVeg"
436    _adapter: WIEAdapter
437    factorial: str | None = None  # per-model factorial name, e.g. "baseline", "ndep"
438    invalid_combination = False
439
440    @property
441    def kind(self) -> str:
442        """Is the variable a stock or flux? Stocks don't carry a s-1 unit and as such are not multiplied by seconds."""
443        return kind_of(self.variable)
444
445    @property
446    def units(self) -> str:
447        """Native units string from the file header."""
448        return self.read().attrs.get("units", "")
449
450    @property
451    def path(self) -> str:
452        """Resolved bucket path, delegated to the model's adapter."""
453        return str(
454            self._adapter.path(
455                self.experiment,
456                self.simulation,
457                self.forcing,
458                self.factorial,
459                self.variable,
460            )
461        )
462
463    @property
464    def paths(self) -> list[str]:
465        """Every file behind this variable's series (one, unless the model chunked it)."""
466        return [
467            str(p)
468            for p in self._adapter.paths(
469                self.experiment,
470                self.simulation,
471                self.forcing,
472                self.factorial,
473                self.variable,
474            )
475        ]
476
477    def read(self) -> xr.DataArray:
478        """Standardized, *lazy* DataArray for this variable (canonical dims,
479        pandas-datetime time coord at native cadence, NaN fills, native units).
480
481        Raises whatever opening the file raises (FileNotFoundError for a combo
482        that wasn't uploaded): read() is the single source of truth for what
483        exists, so the caller can catch and report it. path() never pre-judges.
484        """
485
486        if not self.exists():
487            if self.invalid_combination:
488                raise InvalidSimulationError(
489                    f"The combination of {self.experiment}, {self.simulation}, {self.forcing}, "
490                    f"{self.factorial}, and {self.variable} does not exist in the WIEMIP protocol."
491                )
492            else:
493                pth = self._adapter.path(
494                    self.experiment,
495                    self.simulation,
496                    self.forcing,
497                    self.factorial,
498                    self.variable,
499                )
500                raise FileNotFoundError(
501                    f"Missing file for {pth}. Requested: {self.experiment}, {self.simulation}, {self.forcing}, {self.factorial}, {self.variable}"
502                )
503
504        return self._adapter.read(
505            self.experiment,
506            self.simulation,
507            self.forcing,
508            self.factorial,
509            self.variable,
510        )
511
512    def weighted_dataarray(
513        self, da: xr.DataArray | None = None
514    ) -> xr.core.weighted.DataArrayWeighted:
515        """Wrap the data in this model's documented area weights, so a sum over
516        (lat, lon) integrates the per-m2 quantity. Delegated to the adapter."""
517        if da is None:
518            da = self.read()
519        return self._adapter.weight_dataarray(da)
520
521    @ensure_valid
522    def exists(self):
523        """Does the file exist? Also returns false if a model hasn't been implemented yet in the registry."""
524        try:
525            pths = self.paths
526        except (MissingFactorialError, NotImplementedError):
527            # No such factorial for this model, or its overshoot naming isn't mapped
528            # yet: either way there is nothing on disk to find. exists() is the
529            # non-raising probe, so callers can sweep a request set without guarding
530            # every combo; read() stays the gate that raises.
531            return False
532        return all(os.path.isfile(p) for p in pths)
533
534    @cache_csv
535    def latitudinal_sum(
536        self,
537        start: float | None = None,
538        end: float | None = None,
539    ) -> pd.Series:
540        """Area-weighted total as a Pg C series at the file's native cadence
541        (monthly stays monthly). With no band, sums the whole globe; pass
542        (start, end) degrees to restrict to a latitude band. The unit conversion is
543        delegated to the model's adapter (`to_pgc`).
544
545        Wrapped by `@cache_csv`: the result is mirrored to a CSV under
546        `const.CSV_ROOT` and reused until the source .nc is newer or overwrite is True.
547        """
548        da = self.read()
549        if start is not None and end is not None:
550            band = da.sel(lat=slice(start, end))
551            if band.sizes.get("lat", 0) == 0:  # handle descending-lat grids
552                band = da.sel(lat=slice(end, start))
553            da = band
554        total = self.weighted_dataarray(da).sum(("lat", "lon"))
555        return self._adapter.to_pgc(total, self.variable)
556
557    def __repr__(self) -> str:
558        if self.factorial is None:
559            return (
560                f"WIEFile({self.experiment}.{self.simulation}.{self.model}."
561                f"{self.forcing}.{self.variable})"
562            )
563        else:
564            return (
565                f"WIEFile({self.experiment}.{self.simulation}.{self.model}."
566                f"{self.forcing}.{self.factorial}.{self.variable})"
567            )

 73    Hand-deny any strange combination of arguments.

 75    accept unconventional arguments (e.g., UKESM forcing a ctrl simulation),

 77    So:
 78     bgc/ctrl should only return True for stable climate
 79     cou should only return True for ukesm/gfdl/ipsl

 84    def wrapper(self):
 85
 86        if self.experiment == const.ONE_PERCENT_CO2_KEY:
 87            # cou and rad both include the radiative effect of CO2, so they are
 88            # driven with a transient GCM-pattern climate. bgc and ctrl hold the
 89            # climate constant, so they take the stable pattern.
 90            if self.simulation in (
 91                const.OnePctSimulation.cou.name,
 92                const.OnePctSimulation.cou_ndep.name,
 93                const.OnePctSimulation.rad.name,
 94                const.OnePctSimulation.rad_ndep.name,
 95            ):
 96                if self.forcing not in (
 97                    const.GCMPattern.ukesm.name,
 98                    const.GCMPattern.ipsl.name,
 99                    const.GCMPattern.gfdl.name,
100                ):

102                    return False
103            elif self.simulation in (
104                const.OnePctSimulation.bgc.name,
105                const.OnePctSimulation.bgc_ndep.name,
106            ):

def spherical_area( obj: xarray.core.dataset.Dataset | xarray.core.dataarray.DataArray, latn: str, lonn: str) -> xarray.core.dataarray.DataArray: 
294def spherical_area(
295    obj: xr.Dataset | xr.DataArray, latn: str, lonn: str
296) -> xr.DataArray:
297    """
298    Grid-cell area [m2] from lat/lon centres, assuming a regular spherical grid.
299
300    Used by the models that do *not* ship an area raster (CLASSIC, DLEM, JSBACH,
301    JULES cell, VISIT-UT). Models with a provided raster (BiomeE `veg_area.nc`,
302    LPX-Bern `gridcell_area.nc`) ignore this. Verbatim from extract.py.
303    """
304    lat, lon = obj[latn].values, obj[lonn].values
305    R = 6.371e6
306    dlat, dlon = np.abs(np.gradient(lat)), np.abs(np.gradient(lon))
307    band = R**2 * (
308        np.sin(np.deg2rad(lat + dlat / 2)) - np.sin(np.deg2rad(lat - dlat / 2))
309    )
310    return xr.DataArray(
311        band[:, None] * np.deg2rad(dlon)[None, :],
312        dims=(latn, lonn),
313        coords={latn: obj[latn], lonn: obj[lonn]},
314    ).astype("float32")

332def standardize(
333    da: xr.DataArray, latn: str, lonn: str, time: np.ndarray
334) -> xr.DataArray:

370def years_to_datetime(values) -> np.ndarray:
371    """Numeric (possibly fractional) *calendar* years -> `datetime64[M]`, keeping
372    sub-annual resolution: year = floor(v), month = round(frac * 12) clamped 0..11.
373    Annual data (frac == 0) maps to January of each year. Used by the models whose
374    time axis is a bare numeric year (LPX-Bern, VISIT-UT)."""
375    v = np.asarray(values, dtype="float64")

377    months = np.clip(np.rint((v - years) * 12).astype("int64"), 0, 11)
378    total_months = (years - 1970) * 12 + months  # months since the 1970 epoch
379    return np.datetime64("1970-01", "M") + total_months.astype("timedelta64[M]")

394    only when that CSV is missing or older than the source variable file; on a hit
395    it reads the CSV straight back (two stats, zero netCDF reads).
396
397    Freshness keys on the *variable* file (`self.path`) only.
398    Use overwrite to recompute the sum. Useful when cache is invalid or methods change.
399    """
400
401    @functools.wraps(method)
402    def wrapper(self, start=None, end=None, overwrite=False):
403        srcs = [Path(p) for p in self.paths]  # pure transform == what read() opens
404        out = _csv_path(srcs[0], start, end)
405        if (
406            not overwrite
407            and out.exists()
408            and all(src.exists() for src in srcs)
409            and out.stat().st_mtime >= max(src.stat().st_mtime for src in srcs)
410        ):
411            series = pd.read_csv(out, index_col=0, parse_dates=True).iloc[:, 0]
412            # read_csv picks a resolution per file, so a series that stops before 2262
413            # comes back in ns and one running to 2300 in us; adding the two then
414            # overflows. Pin both to what standardize() hands out.
415            series.index = series.index.astype("datetime64[us]")
416            return series
417        series = method(self, start, end)
418        out.parent.mkdir(parents=True, exist_ok=True)
419        series.to_csv(out)
420        return series
421
422    return wrapper
